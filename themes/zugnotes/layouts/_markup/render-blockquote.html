with a soft "\n", so the date line and the explanation ran together. 📅" -}} {{- $class = "dated-note" -}} {{- if findRE `^

📅[^\n<]*\n` $text 1 -}} {{- $text = replaceRE `^

(📅[^\n<]*?)[ \t]*\n` "

$1

\n

" $text 1 -}} {{- else -}} {{- $text = replaceRE `^

📅` "

📅" $text 1 -}} {{- end -}} {{- end -}} {{- $text = replaceRE `\n(—\s)` "

\n

$1" $text -}}
